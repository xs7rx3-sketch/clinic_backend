business_rules:
  appointment_overlap: |
    Two appointments conflict if ALL of the following hold:
      - existing.start_at < requested_end
      - existing.end_at > requested_start
      - existing.status NOT IN ('CANCELLED', 'NO_SHOW', 'RESCHEDULED_BY_VIP')
    Always exclude the appointment being modified (if any) via: id != :current_id

  vip_window_lockdown: |
    When securing a department for a VIP delegation:
      1. Identify all overlapping appointments in that department (status NOT IN CANCELLED/NO_SHOW, is_vip_booking=FALSE).
      2. Reschedule each by +3 hours (or next available slot).
      3. Log each move in displaced_appointment_log with bilingual notification_ar and notification_en.
      4. Insert one row in vip_delegations with status='ACTIVE_LOCKDOWN'.
      5. Insert one VIP appointment in appointments with is_vip_booking=TRUE, booking_source='VIP_LIAISON'.
